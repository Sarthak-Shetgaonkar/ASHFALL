cmake_minimum_required(VERSION 3.10)
project(Ashfall CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
find_package(SDL2 CONFIG REQUIRED)
add_executable(ashfall src/main.cpp)
# Link only the core SDL2 library (the code uses SDL_MAIN_HANDLED, so SDL2main is not needed)
if(TARGET SDL2::SDL2)
  target_link_libraries(ashfall PRIVATE SDL2::SDL2)
else()
  target_include_directories(ashfall PRIVATE ${SDL2_INCLUDE_DIRS})
  target_link_libraries(ashfall PRIVATE ${SDL2_LIBRARIES})
endif()
if(UNIX)
  target_link_libraries(ashfall PRIVATE m)
endif()
if(MSVC)
  target_compile_options(ashfall PRIVATE /O2 /utf-8)
  target_compile_definitions(ashfall PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
